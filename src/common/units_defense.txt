# Mass/Force relations
g0 = 9.80665 * meter / second ** 2 = standard_gravity
slug = pound_force * second ** 2 / foot
lbm = pound = avoirdupois_pound
lbf = pound_force
kip = 1000 * pound_force

# Pressure convenience
psi = pound_force / inch ** 2
ksi = kip / inch ** 2
psf = pound_force / foot ** 2
inH2O = 249.08891 * pascal  # 4°C approx reference
inHg = 3386.38816 * pascal  # 0°C approx reference

# Speed & distance
nautical_mile = 1852 * meter = nmi
knot = nautical_mile / hour = kn
mil = 0.001 * inch

# Temperature aliases
degF = degFahrenheit = deg_F
degC = degCelsius = deg_C
